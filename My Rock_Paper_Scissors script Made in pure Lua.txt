math.randomseed(os.time())

local PickTable = {"Rock 🪨", "Paper 📄", "Scissors ✂️"}
local WinForUser, WinForCom, Draw = 0,0,0
local GameRounds = 0

local WinLines = {
    "Sheesh, human actually cooked! 🔥",
    "Nah you got lucky, run it back!",
    "Aura +1000 for you, not bad."
}
local LoseLines = {
    "LOL lost to scraps 💀",
    "My grandma plays better and she's a toaster",
    "AI Aura farms - Now it's time to join the Honored One"
}

local function ComputerBrain()
    return PickTable[math.random(#PickTable)]
end

print("=== ROCK PAPER SCISSORS - Best of 5 ===\nChoose: 1=Rock, 2=Paper, 3=Scissors\n")

while GameRounds < 5 do
    io.write("\nRound "..(GameRounds+1).."/5 - Enter 1-3: ")
    local UserInput = tonumber(io.read())
    local InvalidityCounter = 0

    while not UserInput or UserInput < 1 or UserInput > 3 do
        InvalidityCounter = InvalidityCounter + 1
        if InvalidityCounter == 3 then
            print("Bro stop gooning the code 😭")
        elseif InvalidityCounter >= 5 then
            print("Aight you win the gooning award, you lose automatically. Bye.")
            WinForCom = 5
            GameRounds = 5
            break
        end
        io.write("Invalid! Try again 1-3: ")
        UserInput = tonumber(io.read())
    end

    if GameRounds == 5 then break end

    local UserChoice = PickTable[UserInput]
    local ComChoice = ComputerBrain()

    print("You: "..UserChoice.." vs Me: "..ComChoice)
    GameRounds = GameRounds + 1

    -- Draw
    if UserChoice == ComChoice then
        Draw = Draw + 1
        print("Draw! Like minds think alike... we lose alike 😏")
    -- Win conditions (strip emojis for logic)
    elseif (UserInput == 1 and ComChoice:find("Scissors")) or
           (UserInput == 2 and ComChoice:find("Rock")) or
           (UserInput == 3 and ComChoice:find("Paper")) then
        WinForUser = WinForUser + 1
        print("You win this round! "..WinLines[math.random(#WinLines)])
    else
        WinForCom = WinForCom + 1
        print("I win! "..LoseLines[math.random(#LoseLines)])
    end

    print("Score: You "..WinForUser.." - "..WinForCom.." Me | Draws: "..Draw)

    if WinForUser == 3 or WinForCom == 3 then
        break -- first to 3 wins early
    end
end

print("\n=== FINAL ===")
print("You: "..WinForUser.." | Me: "..WinForCom.." | Draws: "..Draw)

if WinForUser > WinForCom then
    print("The Power of Human Intellect overthrew the AI 👑 GG!")
elseif WinForCom > WinForUser then
    print("GGs humanoid, still lost. AI whispers ('Here's your 100 Krypt') 💸")
else
    print("Draw after "..GameRounds.." rounds?! We are the same fr.")
end
